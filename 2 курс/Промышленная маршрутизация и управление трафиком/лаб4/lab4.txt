####### HKI ########
/interface bridge add name=lo0
/ip address
add address=10.0.0.11/32 interface=lo0
add address=10.11.21.1/30 interface=ether2
add address=10.11.31.1/30 interface=ether3
add address=10.11.10.1/30 interface=ether4
/routing ospf instance set default router-id=10.0.0.11
/routing ospf network
add network=10.0.0.11/32 area=backbone
add network=10.11.21.0/30 area=backbone
add network=10.11.31.0/30 area=backbone
add network=10.11.10.0/30 area=backbone
/mpls ldp set enabled=yes transport-address=10.0.0.11 lsr-id=10.0.0.11
/mpls ldp interface
add interface=ether2
add interface=ether3
add interface=ether4
/routing bgp instance set default as=65000 router-id=10.0.0.11 redistribute-connected=yes
/routing bgp peer
add name=to_LND remote-address=10.0.0.21 remote-as=65000 update-source=lo0 address-families=vpnv4
add name=to_LBN remote-address=10.0.0.31 remote-as=65000 update-source=lo0 address-families=vpnv4
add name=to_SPB remote-address=10.0.0.10 remote-as=65000 update-source=lo0 route-reflect=yes address-families=vpnv4
